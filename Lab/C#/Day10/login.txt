static void Main(string[] args) { // Seed users (like a database) List<User> users = new List<User> { new User { Username = "admin", Password = "1234" }, new User { Username = "john", Password = "abcd" }, new User { Username = "jane", Password = "pass" } }; Console.WriteLine("=== Login System ==="); // Input credentials Console.Write("Enter Username: "); string inputUsername = Console.ReadLine(); Console.Write("Enter Password: "); string inputPassword = Console.ReadLine(); // LINQ query to check credentials var matchedUser = users .Where(u => u.Username == inputUsername && u.Password == inputPassword) .FirstOrDefault(); // Result if (matchedUser != null) { Console.WriteLine("✅ Login Successful! Welcome " + matchedUser.Username); } else { Console.WriteLine("❌ Invalid Username or Password."); } Console.ReadKey(); }
