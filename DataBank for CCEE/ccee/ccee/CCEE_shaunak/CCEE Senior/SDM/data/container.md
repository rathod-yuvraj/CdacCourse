## container

```bash

# stop the container
> docker container stop <container name or id>

# run a container in detached mode
> docker container run -d --name <container name> <image name or id>
> docker container run -d --name server httpd

# remove a container forcefully
# first the container gets stopped (implicitly)
# and then the container gets removed
> docker container rm --force <container name or id>

# run container in interactive mode
# run a container to interact with it (in future)
# -i: interactive
# -t: get the teletype terminal while interacting with container
> docker container run -d -i -t --name <container name> <image name or id>
> docker container run -itd --name <container name> <image name or id>

# execute a command inside the container and get the output
> docker container exec -it <container name or id> <command>
> docker container exec -it myubuntu date

# get shell of running container
> docker container exec -it <container name or id> bash
> docker container exec -it <container name or id> sh

# get container details
> docker container inspect <container id or name>

# run a container with port forwarding
> docker container run -itd --name <container name> -p <source port>:<container port> <image name or id>
> docker container run -itd --name myhttpd -p 8090:80 httpd

# run a container and restart it always (even after crashing the inner program or restarting the machine)
> docker container run -itd --name <container name> -p <source port>:<container port> --restart always <image name>

# read the logs generated by container
> docker container logs <container id or name

```
