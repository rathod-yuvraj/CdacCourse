#include<iostream>
#include<string.h>
#include<malloc.h>
using namespace std;
class Account
{
private:
	char *id,*name;
	int balance;
	const int rate;
public:
	Account(char *id,char *name,int balance,int rt):rate(rt)
	{
		this->id=new char[strlen(id)+1];
		strcpy(this->id,id);

		this->name=new char[strlen(name)+1];
		strcpy(this->name,name);
		this->balance=balance;

	}
	Account():rate(3)
	{
		id=NULL;
		name=NULL;
		balance=0;
	}
	void setId(char *id)
	{
		this->id=new char[strlen(id)+1];
		strcpy(this->id,id);
	}
	void setName(char *name)
	{
		this->name=new char[strlen(name)+1];
		strcpy(this->name,name);
	}
	void setBalance(int balance)
	{
		this->balance=balance;
	}
	char* getId()
	{
		static char *sp=" ";
		if(id!=NULL)
		{
			return id;
		}
		else
		{
			return sp;
		}
	}
	char* getName()
	{
		static char *sp=" ";
		if(name!=NULL)
		{
			return name;
		}
		else
		{
			//return "hello";		
			return sp;
		}
	}
	int getBalance()
	{
		return balance;
	}
	Account(Account &ref):rate(ref.rate)
	{
		if(ref.id!=NULL)
		{
			id=new char[strlen(ref.id)+1];
			strcpy(id,ref.id);
		}
		else
		{
			id=ref.id;
		}

		if(ref.name!=NULL)
		{
			name=new char[strlen(ref.name)+1];
			strcpy(name,ref.name);
		}
		else
		{
			name=ref.name;
		}
		balance=ref.balance;
	}
		~Account()
		{
			cout<<"Inside destructor"<<endl;
			if(id!=NULL)
			{
				delete []id;
			}

			if(name!=NULL)
			{
				delete []name;
			}
		}
};
